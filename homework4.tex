#include <iostream>
#include <vector>

using namespace std;

// Expoenitnation
long long modPow(long long base, long long exp, long long mod) {
    long long res = 1;
    base %= mod;
    while (exp > 0) {
        if (exp % 2 == 1) res = (res * base) % mod;
        base = (base * base) % mod;
        exp /= 2;
    }
    return res;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int e = 84733;
    int n = 89951;

    // Sieve of Eratosthenes
    vector<bool> stiv(n + 1, false);
    vector<int> primes;

    for (int i = 2; i <= n; i++) {
        if (!stiv[i]) {
            primes.push_back(i);
            for (int j = i * 2; j <= n; j += i) {
                stiv[j] = true;
            }
        }
    }

    // Find prime factors p and q
    long long p = 0, q = 0;
    int len = primes.size();
    for (int i = len - 1; i >= 0; i--) {
        for (int j = 0; j < len; j++) {
            long long prod = (long long)primes[i] * primes[j];
            if (prod == n) {
                p = primes[j];
                q = primes[i];
                i = -1; // break outer loop
                break;
            }
            else if (prod > n) {
                break;
            }
        }
    }

    long long phi = (p - 1) * (q - 1);

    // Find private exponent d
    long long d = -1;
    for (long long i = 1; i <= phi; i++) {
        if ((i * e) % phi == 1) {
            d = i;
            break;
        }
    }

    long long c;
    while (cin >> c) {
        cout << modPow(c, d, n) << "\n";
    }

    return 0;

Let $d=\gcd(a,b)=ax_0+by_0$ and $g=\gcd(ka,kb)$. Then $kd=ka\,x_0+kb\,y_0$, so $g\mid kd$.
Also $d\mid a,\ d\mid b \Rightarrow kd\mid ka,\ kd\mid kb \Rightarrow kd\mid g$.
Hence $g=kd$.
\end{proof}
\item
\begin{proof}

    \[\gcd(a, c) = ax_2+cy_2 = 1\]

    Because there exist a combination of integers $P$, $Q$ such that $aP+bcQ=1$ them $\gcd(a, bc)=1$
    \end{proof}

#include <iostream>
#include <sstream>
#include <string>
#include <cctype>
using namespace std;

const string RED = "\033[31m";
const string RESET_COLOR = "\033[0m";

bool locked[256] = { false };

// Print the string, showing locked letters in red
void printString(const string& s) {
    for (char c : s) {
        if (locked[(unsigned char)c])
            cout << RED << c << RESET_COLOR;
        else
            cout << c;
    }
    cout << endl;
}

void clearLocks() {
    for (int i = 0; i < 256; i++) locked[i] = false;
}

int main() {
    string original, current;

    original = "FG LTEKTZL STYZ YGK TBETHZOGFQS LZXRTFZL YKGD LGYZVQKT TFUOFTTKOFU AFGVF QL YQY YKGD XZD";
    current = original;

    cout << "Commands:\n"
        << "  A B        swap the letters A and B\n"
        << "  lock A B T lock letters (shown in red, can't be swapped)\n"
        << "  unlock     unlock all letters\n"
        << "  reset      restore the original string (and unlock everything)\n"
        << "  quit       exit\n";

    string line;
    while (cout << "> " && getline(cin, line)) {
        istringstream iss(line);
        string cmd;
        if (!(iss >> cmd)) continue;

        string lower = cmd;
        for (char& c : lower) c = tolower((unsigned char)c);

        if (lower == "quit") {
            break;
        }

        if (lower == "reset") {
            current = original;
            clearLocks();
            printString(current);
            continue;
        }

        if (lower == "unlock") {
            clearLocks();
            printString(current);
            continue;
        }

        if (lower == "lock") {
            string letters;
            while (iss >> letters) {
                locked[(unsigned char)letters[0]] = true;
            }
            printString(current);
            continue;
        }

        // Otherwise: swap command "A B"
        string second;
        if (!(iss >> second)) {
            cout << "Please type two letters, e.g. A B\n";
            continue;
        }

        char a = cmd[0], b = second[0];

        if (locked[(unsigned char)a] || locked[(unsigned char)b]) {
            cout << "Can't swap: "
                << (locked[(unsigned char)a] ? a : b)
                << " is locked.\n";
            printString(current);
            continue;
        }

        for (char& c : current) {
            if (c == a)      c = b;
            else if (c == b) c = a;
        }

        printString(current);
    }

    return 0;
}
