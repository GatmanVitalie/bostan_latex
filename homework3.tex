\section*{Problem 3.3}


\[
1 + \frac{1}{4} + \frac{1}{9} + \frac{1}{16} + \dots + \frac{1}{n^2} < 2 - \frac{1}{n}, \quad \forall n \in \mathbb{N}, \, n > 1.
\]

\begin{proof}
Lets us define: $\displaystyle f(n)= \sum_{i=1}^{n} \frac{1}{i^2}$.

\paragraph{1. Base case ($n = 2$):}
For $n = 2$, if we evaluate both sides we get:
\[
\text{LHS} = f(2) = 1 + \frac{1}{4} = \frac{5}{4} = 1.25, \quad \text{RHS} = 2 - \frac{1}{2} = \frac{3}{2} = 1.5.
\]
Since $1.25 < 1.5$, the inequality holds true.

\paragraph{2. We prove the induction}
\[
1 + \frac{1}{4} + \frac{1}{9} + \dots + \frac{1}{k^2} < 2 - \frac{1}{k}.
\]
\[
1 + \frac{1}{4} + \dots + \frac{1}{k^2} + \frac{1}{(k+1)^2} < 2 - \frac{1}{k} + \frac{1}{(k+1)^2}.
\]
\[
2 - \frac{1}{k} + \frac{1}{(k+1)^2} \le 2 - \frac{1}{k+1} \iff \frac{1}{(k+1)^2} \le \frac{1}{k} - \frac{1}{k+1}.
\]
\[
\frac{1}{k} - \frac{1}{k+1} = \frac{(k+1) - k}{k(k+1)} = \frac{1}{k(k+1)}.
\]
\[
\frac{1}{(k+1)^2} < \frac{1}{k(k+1)} = \frac{1}{k} - \frac{1}{k+1}.
\]
\[
1 + \frac{1}{4} + \dots + \frac{1}{(k+1)^2} < 2 - \frac{1}{k} + \left(\frac{1}{k} - \frac{1}{k+1}\right) = 2 - \frac{1}{k+1}.
\]
\end{proof}

\section*{Problem 3.4}
\[
F_1 + F_2 + F_3 + \dots + F_n = F_{n+2} - 1, \quad \forall n \in \mathbb{N}, \, n \ge 1.
\]

\begin{proof}
From the definition of the Fibonacci sequence: $F_1 = 1$, $F_2 = 1$, and $F_k = F_{k-1} + F_{k-2}$ for all integers $k \ge 3$. 

We define be the statement: $\displaystyle f(n) = \sum_{i=1}^{n} F_i = F_{n+2} - 1$.

\paragraph{1. Base Case ($n = 1$):}
For $n = 1$, if we evaluate both sides:
\[
\text{LHS} = f(1) = F_1 = 1, \quad \text{RHS} = F_{1+2} - 1 = F_3 - 1 = 2 - 1 = 1.
\]
Since $\text{LHS} = \text{RHS}$, the inequiality holds true.

\paragraph{2. We prove the induction}
\[
F_1 + F_2 + F_3 + \dots + F_k = F_{k+2} - 1.
\]
\[
F_1 + F_2 + \dots + F_k + F_{k+1} = F_{(k+1)+2} - 1 = F_{k+3} - 1.
\]
\begin{align*}
\sum_{i=1}^{k+1} F_i &= \left( \sum_{i=1}^{k} F_i \right) + F_{k+1} \\
&= (F_{k+2} - 1) + F_{k+1}  \\
&= (F_{k+2} + F_{k+1}) - 1.
\end{align*}
We can rewrite $F_{k+3}$ as, $F_{k+3} = F_{k+2} + F_{k+1}$. Substituting this in:
\[
\sum_{i=1}^{k+1} F_i = F_{k+3} - 1.
\]
Thus the equality holds true.
\end{proof}

\[
4^k + 7 \le 5^k.

4^{k+1} + 7 \le 5^{k+1}.

Thus, the inequality holds.
\end{proof}
\section*{Problem 3.6}

A positive integer $n$ is called a \textbf{perfect number} if it is equal to the sum of its proper positive divisors.
Prove by induction that if $2^k - 1$ is a prime number (a \textbf{Mersenne prime}), then $n = 2^{k-1}(2^k - 1)$ is an even perfect number.

\begin{proof}
Let $p = 2^k - 1$ be a Mersenne prime. Since $p$ is prime and $p > 1$, we must have $k \ge 2$, so $n = 2^{k-1}p$ is divisible by $2$ and is therefore even.

The proper positive divisors of $n = 2^{k-1}p$ consist of:
\begin{enumerate}
    \item Divisors that are powers of 2: $1, 2, 2^2, \dots, 2^{k-1}$.
    \item Divisors multiplied by $p$ (excluding $n$ itself): $p, 2p, 2^2p, \dots, 2^{k-2}p$.
\end{enumerate}

Thus, the sum of all proper positive divisors of $n$ is:
\[
S = \sum_{i=0}^{k-1} 2^i + p \sum_{i=0}^{k-2} 2^i.
\]

We first prove by induction that for any integer $m \ge 0$:
\[
g(m) = \sum_{i=0}^{m} 2^i = 2^{m+1} - 1.
\]

\paragraph{1. Base Case ($m = 0$):}
For $m = 0$:
\[
\text{LHS} = g(0) = 2^0 = 1, \quad \text{RHS} = 2^{0+1} - 1 = 2 - 1 = 1.
\]

\paragraph{2. We prove the induction}
\[
\sum_{i=0}^{j+1} 2^i = \left(\sum_{i=0}^{j} 2^i\right) + 2^{j+1} = (2^{j+1} - 1) + 2^{j+1} = 2 \cdot 2^{j+1} - 1 = 2^{j+2} - 1.
\]
Thus, the sum formula holds for all $m \ge 0$.

Using $g(k-1)$ and $g(k-2)$:
\[
\sum_{i=0}^{k-1} 2^i = 2^k - 1 = p, \quad \sum_{i=0}^{k-2} 2^i = 2^{k-1} - 1.
\]
\[
S = p + p(2^{k-1} - 1) = p\left(1 + 2^{k-1} - 1\right) = p \cdot 2^{k-1} = n.
\]
Since the sum of the proper positive divisors of $n$ equals $n$, $n$ is a perfect number. Since $k \ge 2$, $n$ is an even perfect number.
\end{proof}


\section*{Problem 3.7}
\begin{proof}
    \[
F_n < 2^n.
\]
\paragraph{1. Base Cases ($n = 1$ and $n = 2$):}
For $n = 1$:
\[
\text{LHS} = F_1 = 1, \quad \text{RHS} = 2^1 = 2.
\]
Since $1 < 2$, the inequality holds true.

\vspace{0.5em}
\noindent For $n = 2$:
\[
\text{LHS} = F_2 = 1, \quad \text{RHS} = 2^2 = 4.
\]
Since $1 < 4$, the inequality holds true.

\paragraph{2. We prove the induction}
Assume by strong induction that $F_i < 2^i$ holds for all $1 \le i \le k$, where $k \ge 2$. We show that $F_{k+1} < 2^{k+1}$.

By definition of the Fibonacci sequence:
\[
F_{k+1} = F_k + F_{k-1}.
\]
Applying the strong inductive hypothesis to $F_k$ and $F_{k-1}$ (since $1 \le k-1 < k$):
\[
F_k < 2^k \quad \text{and} \quad F_{k-1} < 2^{k-1}.
\]
Substituting these into the expression for $F_{k+1}$:
\[
F_{k+1} = F_k + F_{k-1} < 2^k + 2^{k-1}.
\]
Since $2^{k-1} < 2^k$ for all $k \ge 2$:
\[
2^k + 2^{k-1} < 2^k + 2^k = 2 \cdot 2^k = 2^{k+1}.
\]
Therefore:
\[
F_{k+1} < 2^{k+1}.
\]
Thus, the inequality holds.
\end{proof}
