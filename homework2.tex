\begin{proof}
Contrapositive: if $r^{1/5}$ is rational, then $r$ is rational.
Assume that $r^{1/5}$ is rational. Then there exist $m,n \in \mathbb{N}$ with
\[
r^{1/5} = \frac{m}{n}.
\]
Therefore,
\[
r = \left(r^{1/5}\right)^5 = \left(\frac{m}{n}\right)^5,
\]
which is rational. Thus, the contrapositive is true, and consequently, if $r$ is irrational, then $r^{1/5}$ is irrational.
\end{proof}

Since $(\beta \land \neg\beta)$ is a contradiction, its truth value is always false ($\text{F}$).

\begin{itemize}
    \item If $\alpha$ is true ($\text{T}$), then $\alpha \to (\beta \land \neg\beta) \equiv \text{T} \to \text{F} \equiv \text{F}$.
    \item If $\alpha$ is false ($\text{F}$), then $\alpha \to (\beta \land \neg\beta) \equiv \text{F} \to \text{F} \equiv \text{T}$.
\end{itemize}

In both cases, $\alpha \to (\beta \land \neg\beta)$ yields the same truth value as $\neg\alpha$. Therefore, the two formulas are logically equivalent.

\section*{Problem 2.8}
Using De Morgan's laws:
\[
P \equiv (A \lor B) \to C \equiv \neg(A \lor B) \lor C \equiv (\neg A \land \neg B) \lor C
\]
Distributing $\lor C$:
\[
P \equiv (\neg A \lor C) \land (\neg B \lor C) \equiv (A \to C) \land (B \to C)
\]

We simplify $Q$ using contrapositives ($\neg C \to \neg X \equiv X \to C$):
\[
Q \equiv (\neg C \to \neg A) \lor (\neg C \to \neg B) \equiv (A \to C) \lor (B \to C)
\]

Since conjunction implies disjunction ($X \land Y \implies X \lor Y$), we have:
\[
((A \to C) \land (B \to C)) \implies ((A \to C) \lor (B \to C))
\]
Therefore, $P \to Q$ holds, but $P$ and $Q$ are not logically equivalent.

The converse fails: take $A=\text{F}$, $B=\text{T}$, $C=\text{F}$. Then $A \to C$ is true,
so $Q$ is true, but $B \to C$ is false, so $P$ is false. Hence $Q \not\to P$
and $P \not\equiv Q$.

\textbf{Answer:} \textbf{b) $P \to Q$.}

\section*{Problem 2.11}

\subsection*{1. Proof of Contrapositive Rule}
We know that $(\alpha \to \beta) \equiv (\neg\beta \to \neg\alpha)$ and using logical equivalences:
\begin{align*}
\alpha \to \beta &\equiv \neg\alpha \lor \beta && \text{(Conditional Law)} \\
&\equiv \beta \lor \neg\alpha && \text{(Commutativity)} \\
&\equiv \neg(\neg\beta) \lor \neg\alpha && \text{(Double Negation)} \\
&\equiv \neg\beta \to \neg\alpha && \text{(Conditional Law)}
\end{align*}
Because $\alpha \to \beta \equiv \neg\beta \to \neg\alpha$, the premise logically implies the conclusion:
\[
(\alpha \to \beta) \models (\neg\beta \to \neg\alpha)
\]

\subsection*{2. Proof of Composed Contrapositive Rule}
\begin{align*}
(\alpha \land \beta) \to \gamma 
&\equiv \neg(\alpha \land \beta) \lor \gamma && \text{(Conditional Law)} \\
&\equiv (\neg\alpha \lor \neg\beta) \lor \gamma && \text{(De Morgan's Law)} \\
&\equiv \neg\alpha \lor (\neg\beta \lor \gamma) && \text{(Associativity)} \\
&\equiv \neg\alpha \lor (\gamma \lor \neg\beta) && \text{(Commutativity)} \\
&\equiv (\neg\alpha \lor \gamma) \lor \neg\beta && \text{(Associativity)} \\
&\equiv \neg(\alpha \land \neg\gamma) \lor \neg\beta && \text{(De Morgan's Law: } \neg(A \land \neg B) \equiv \neg A \lor B\text{)} \\
&\equiv (\alpha \land \neg\gamma) \to \neg\beta && \text{(Conditional Law)}
\end{align*}

Because $(\alpha \land \beta) \to \gamma \equiv (\alpha \land \neg\gamma) \to \neg\beta$, the deduction holds:
\[
((\alpha \land \beta) \to \gamma) \models ((\alpha \land \neg\gamma) \to \neg\beta)
\]

Let the propositions be defined as:
\begin{align*}
S &: \text{Smith is the murderer.} \\
L &: \text{Jones is lying.} \\
V &: \text{Jones has seen Smith last night.} \\
M &: \text{The murder happened after midnight.}
\end{align*}
\newpage
The premises are formalized as:
\begin{enumerate}
    \item $P_1: \neg V \to (S \lor L)$
    \item $P_2: \neg S \to (\neg V \land M)$
    \item $P_3: M \to (S \lor L)$
\end{enumerate}

We test if $P_1 \land P_2 \land P_3 \models S$.

Assume $S$ is false ($S \equiv \text{F}$). From $P_2$, since $\neg S \equiv \text{T}$, we obtain:
\[
\neg V \land M \equiv \text{T} \implies \neg V \equiv \text{T} \text{ and } M \equiv \text{T}.
\]
Substituting $M \equiv \text{T}$ into $P_3$ gives $M \to (S \lor L) \equiv \text{T} \to (\text{F} \lor L) \equiv L$. Thus, for $P_3$ to be true, $L$ must be true ($L \equiv \text{T}$).

Evaluating $P_1$ under this assignment ($\neg V \equiv \text{T}, S \equiv \text{F}, L \equiv \text{T}$):
\[
\neg V \to (S \lor L) \equiv \text{T} \to (\text{F} \lor \text{T}) \equiv \text{T}.
\]

All premises $P_1, P_2, P_3$ evaluate to $\text{T}$ under the valuation $(S=\text{F}, L=\text{T}, V=\text{F}, M=\text{T})$. 

\noindent\textbf{Conclusion:}
The deduction is \textbf{invalid}. We cannot conclude that Smith is the murderer; we can only deduce that $S \lor L$ is true (either Smith is the murderer or Jones is lying).

\begin{itemize}
    \item Box 1: ``The gold is not here''
    \item Box 2: ``The gold is not here''
    \item Box 3: ``The gold is in Box 2''
\end{itemize}

\subsection*{Formalization}

\begin{itemize}
    \item $G_1$: the gold is in Box 1.
    \item $G_2$: the gold is in Box 2.
    \item $G_3$: the gold is in Box 3.
\end{itemize}

\begin{align*}
C_1 &\equiv \neg G_1 && \text{(Box 1: ``the gold is not here'')}\\
C_2 &\equiv \neg G_2 && \text{(Box 2: ``the gold is not here'')}\\

\begin{enumerate}
